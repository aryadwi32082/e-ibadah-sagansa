Kode Google Apps Script (Backend)

  1. Buat Google Sheets baru dengan 3 tab: Siswa, Log_Menstruasi, dan Absensi_Sholat.
  2. Klik menu Ekstensi (Extensions) > Apps Script.
  3. Hapus kode lama, tempel kode di bawah ini, lalu klik Deploy > New Deployment > Web App (Akses: Anyone / Siapa saja).
  4. Salin URL Web App yang muncul dan tempelkan pada kolom di atas.
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetSiswa = ss.getSheetByName("Siswa");
  const sheetAbsen = ss.getSheetByName("Absensi_Sholat");
  const sheetUzur = ss.getSheetByName("Log_Menstruasi");
  
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    siswa: sheetSiswa ? sheetSiswa.getDataRange().getValues() : [],
    absensi: sheetAbsen ? sheetAbsen.getDataRange().getValues() : [],
    uzur: sheetUzur ? sheetUzur.getDataRange().getValues() : []
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const data = JSON.parse(e.postData.contents);
  
  if (data.action === "absen_sholat") {
    let sheet = ss.getSheetByName("Absensi_Sholat");
    if (!sheet) {
      sheet = ss.insertSheet("Absensi_Sholat");
      sheet.appendRow(["ID_Absen", "ID_Siswa", "Nama_Siswa", "Tanggal", "Waktu_Sholat", "Status", "Metode"]);
    }
    sheet.appendRow([
      data.id_absen, data.id_siswa, data.nama_siswa, 
      data.tanggal, data.waktu_sholat, data.status, data.metode
    ]);
  } 
  else if (data.action === "catat_uzur") {
    let sheet = ss.getSheetByName("Log_Menstruasi");
    if (!sheet) {
      sheet = ss.insertSheet("Log_Menstruasi");
      sheet.appendRow(["ID_Log_M", "ID_Siswa", "Nama_Siswa", "Tanggal_Mulai", "Durasi_Hari", "Tanggal_Selesai"]);
    }
    sheet.appendRow([
      data.id_log, data.id_siswa, data.nama_siswa, 
      data.tanggal_mulai, data.durasi, data.tanggal_selesai
    ]);
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: "saved" }))
    .setMimeType(ContentService.MimeType.JSON);
}
